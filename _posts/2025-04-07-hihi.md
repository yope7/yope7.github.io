<script>
const STORAGE_KEY = 'hihiCounterData';
const DATA_VERSION = '1.1';

let counters = {
  battle: 0,
  ring: 0,
  redring: 0,
  eikan: 0,
  hihi: 0
};

let lastDelta = null;

function clampNonNegativeInt(value) {
  const n = Number(value);
  if (!Number.isFinite(n)) return 0;
  return Math.max(0, Math.floor(n));
}

function getBlueboxCount() {
  return (
    clampNonNegativeInt(counters.ring) +
    clampNonNegativeInt(counters.redring) +
    clampNonNegativeInt(counters.eikan) +
    clampNonNegativeInt(counters.hihi)
  );
}

function safeRate(numerator, denominator) {
  if (!denominator || denominator <= 0) return '0.00';
  return ((numerator / denominator) * 100).toFixed(2);
}

function setTextForAll(selector, text) {
  document.querySelectorAll(selector).forEach((el) => {
    el.textContent = String(text);
  });
}

function normalizeLoadedCounters(raw) {
  const c = (raw && raw.counters) ? raw.counters : {};
  return {
    battle: clampNonNegativeInt(c.battle),
    ring: clampNonNegativeInt(c.ring),
    redring: clampNonNegativeInt(c.redring),
    eikan: clampNonNegativeInt(c.eikan),
    hihi: clampNonNegativeInt(c.hihi),
  };
}

// ページ読み込み時にローカルストレージからデータを読み込む
document.addEventListener('DOMContentLoaded', () => {
  const saved = localStorage.getItem(STORAGE_KEY);
  if (saved) {
    try {
      const data = JSON.parse(saved);
      // version 1.0/1.1 を許容（1.0はblueboxを無視し、eikan=0補完）
      if (data && data.counters && (data.version === '1.0' || data.version === '1.1' || !data.version)) {
        counters = normalizeLoadedCounters(data);
      }
    } catch (error) {
      console.error('データの読み込みに失敗しました:', error);
    }
  }
  updateDisplay();
});

function saveToStorage() {
  const data = {
    counters: {
      battle: counters.battle,
      ring: counters.ring,
      redring: counters.redring,
      eikan: counters.eikan,
      hihi: counters.hihi
    },
    timestamp: new Date().toISOString(),
    version: DATA_VERSION
  };
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
}

function exportData() {
  const data = {
    counters: {
      battle: counters.battle,
      ring: counters.ring,
      redring: counters.redring,
      eikan: counters.eikan,
      hihi: counters.hihi
    },
    timestamp: new Date().toISOString(),
    version: DATA_VERSION
  };
  
  const exportArea = document.getElementById('export-area');
  const exportText = document.getElementById('export-text');
  exportText.value = JSON.stringify(data, null, 2);
  exportArea.style.display = 'block';
}

function hideExport() {
  document.getElementById('export-area').style.display = 'none';
}

function copyExport() {
  const exportText = document.getElementById('export-text');
  exportText.select();
  document.execCommand('copy');
}

function showImport() {
  const importArea = document.getElementById('import-area');
  const importText = document.getElementById('import-text');
  importText.value = '';
  importArea.style.display = 'block';
}

function hideImport() {
  document.getElementById('import-area').style.display = 'none';
}

function importData() {
  const importText = document.getElementById('import-text').value;
  if (!importText.trim()) {
    alert('テキストを入力してください。');
    return;
  }

  try {
    const data = JSON.parse(importText);
    if (data && data.counters && (data.version === '1.0' || data.version === '1.1' || !data.version)) {
      counters = normalizeLoadedCounters(data);
      lastDelta = null;
      updateDisplay();
      saveToStorage();
      hideImport();
      alert('データを正常にインポートしました。');
      return;
    }
    alert('無効なデータ形式です。');
  } catch (error) {
    console.error('インポートエラー:', error);
    alert('データのインポートに失敗しました。正しいJSONテキストを入力してください。');
  }
}

function celebrateHihi() {
  // 紙吹雪の演出
  const count = 200;
  const defaults = {
    origin: { y: 0.7 },
    zIndex: 9999
  };

  function fire(particleRatio, opts) {
    confetti({
      ...defaults,
      ...opts,
      particleCount: Math.floor(count * particleRatio),
    });
  }

  fire(0.25, {
    spread: 26,
    startVelocity: 55,
  });
  fire(0.2, {
    spread: 60,
  });
  fire(0.35, {
    spread: 100,
    decay: 0.91,
    scalar: 0.8
  });
  fire(0.1, {
    spread: 120,
    startVelocity: 25,
    decay: 0.92,
    scalar: 1.2
  });
  fire(0.1, {
    spread: 120,
    startVelocity: 45,
  });
}

function applyDelta(delta) {
  const applied = {};
  Object.keys(counters).forEach((key) => {
    const before = counters[key];
    const next = clampNonNegativeInt(before + (delta[key] || 0));
    counters[key] = next;
    applied[key] = next - before;
  });
  lastDelta = applied;
  updateDisplay();
  saveToStorage();
}

function recordBattle() {
  applyDelta({ battle: 1 });
}

function recordDrop(itemKey) {
  if (!['ring', 'redring', 'eikan', 'hihi'].includes(itemKey)) return;
  applyDelta({ battle: 1, [itemKey]: 1 });
  if (itemKey === 'hihi') celebrateHihi();
}

function adjustCounter(key, delta) {
  if (!Object.prototype.hasOwnProperty.call(counters, key)) return;
  applyDelta({ [key]: delta });
}

function undoLast() {
  if (!lastDelta) return;
  const reverse = {};
  Object.keys(lastDelta).forEach((key) => {
    reverse[key] = -lastDelta[key];
  });
  // undo自体はundo対象にしない
  lastDelta = null;
  Object.keys(counters).forEach((key) => {
    counters[key] = clampNonNegativeInt(counters[key] + (reverse[key] || 0));
  });
  updateDisplay();
  saveToStorage();
}

function updateDisplay() {
  const bluebox = getBlueboxCount();

  // カウンター表示の更新（複数箇所まとめて更新）
  setTextForAll('[data-count="battle"]', counters.battle);
  setTextForAll('[data-count="bluebox"]', bluebox);
  setTextForAll('[data-count="ring"]', counters.ring);
  setTextForAll('[data-count="redring"]', counters.redring);
  setTextForAll('[data-count="eikan"]', counters.eikan);
  setTextForAll('[data-count="hihi"]', counters.hihi);

  // 確率計算と表示の更新
  setTextForAll('[data-rate="blueboxRate"]', safeRate(bluebox, counters.battle));
  setTextForAll('[data-rate="ringRate"]', safeRate(counters.ring, bluebox));
  setTextForAll('[data-rate="redringRate"]', safeRate(counters.redring, bluebox));
  setTextForAll('[data-rate="eikanRate"]', safeRate(counters.eikan, bluebox));
  setTextForAll('[data-rate="hihiRate"]', safeRate(counters.hihi, bluebox));

  // 全体ドロップ率（バトル数に対する割合）
  setTextForAll('[data-rate="ringTotalRate"]', safeRate(counters.ring, counters.battle));
  setTextForAll('[data-rate="redringTotalRate"]', safeRate(counters.redring, counters.battle));
  setTextForAll('[data-rate="eikanTotalRate"]', safeRate(counters.eikan, counters.battle));
  setTextForAll('[data-rate="hihiTotalRate"]', safeRate(counters.hihi, counters.battle));

  const undoBtn = document.getElementById('undoBtn');
  if (undoBtn) undoBtn.disabled = !lastDelta;
}

function resetData() {
  if (confirm('本当にすべてのカウンターをリセットしますか？')) {
    counters = {
      battle: 0,
      ring: 0,
      redring: 0,
      eikan: 0,
      hihi: 0
    };
    lastDelta = null;
    updateDisplay();
    saveToStorage();
    alert('すべてのカウンターをリセットしました。');
  }
}
